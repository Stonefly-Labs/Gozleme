using System.Diagnostics;
using Corvus.Text.Json;

var up = args[0];
var docs = new Dictionary<string, byte[]>
{
    ["https://static.modelcontextprotocol.io/schemas/2025-12-11/server.schema.json"] = File.ReadAllBytes(Path.Combine(up, "mcp-registry-2025-12-11.json")),
    ["https://gozleme.invalid/upstream/mcp-2026-07-28.json"] = File.ReadAllBytes(Path.Combine(up, "mcp-2026-07-28.json")),
};
JsonSchemaDocumentResolver resolver = (string uri, out ReadOnlyMemory<byte> json) => { var ok = docs.TryGetValue(uri, out var b); json = b; return ok; };
JsonSchema.Options Offline() => new([], false, JsonSchemaDialect.Draft202012, false, resolver);

void Report(string name, JsonSchema s, string instance)
{
    using var c = JsonSchemaResultsCollector.Create(JsonSchemaResultsLevel.Detailed, 0);
    var ok = s.Validate(instance, c);
    Console.WriteLine($"  [{name}] valid={ok} results={c.GetResultCount()}");
    foreach (var r in c.EnumerateResults())
        if (!r.IsMatch) Console.WriteLine($"    eval={r.GetEvaluationLocationText()} schema={r.GetSchemaEvaluationLocationText()} instance={r.GetDocumentEvaluationLocationText()} msg={r.GetMessageText()}");
}

JsonSchema Compile(string file, string uri)
{
    var sw = Stopwatch.StartNew();
    var s = JsonSchema.FromText(File.ReadAllText(Path.Combine(up, file)), uri, Offline());
    Console.WriteLine($"(a) compile {file}: {sw.Elapsed.TotalMilliseconds:F0} ms");
    return s;
}
